cmake_minimum_required(VERSION 3.24)
project(CaePostExit VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CAE_EXIT_CORE_ONLY "Build the dependency-free lifecycle/ABI tests only" OFF)
include(CTest)
if(CAE_EXIT_CORE_ONLY)
  add_subdirectory(tests)
  return()
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The assigned runtime is x64; a 32-bit plugin is not supported")
endif()
find_package(ParaView 6.2.0 EXACT REQUIRED)
find_package(Qt6 6.9.0 EXACT REQUIRED COMPONENTS Core Gui Widgets)
if(NOT TARGET ParaView::pqApplicationComponents)
  message(FATAL_ERROR "A GUI-enabled ParaView SDK, not the runtime ZIP, is required")
endif()
option(BUILD_SHARED_LIBS "Build the ParaView GUI plugin dynamically" ON)
if(NOT BUILD_SHARED_LIBS)
  message(FATAL_ERROR "Dynamic Qt/plugin linking is required by this delivery")
endif()
if(CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_CONFIGURATION_TYPES "Release" CACHE STRING "Assigned runtime CRT/configuration" FORCE)
elseif(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
  message(FATAL_ERROR "Configure with -DCMAKE_BUILD_TYPE=Release to match the runtime")
endif()
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${CMAKE_INSTALL_BINDIR}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${CMAKE_INSTALL_LIBDIR}")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${CMAKE_INSTALL_LIBDIR}")
set(_paraview_plugin_default_CaePostExit ON)
paraview_plugin_scan(PLUGIN_FILES "${CMAKE_CURRENT_SOURCE_DIR}/Plugin/paraview.plugin"
  PROVIDES_PLUGINS plugins REQUIRES_MODULES required_modules)
foreach(module IN LISTS required_modules)
  if(NOT TARGET "${module}")
    message(FATAL_ERROR "SDK module is missing: ${module}")
  endif()
endforeach()
paraview_plugin_build(RUNTIME_DESTINATION "${CMAKE_INSTALL_BINDIR}"
  LIBRARY_DESTINATION "${CMAKE_INSTALL_LIBDIR}"
  LIBRARY_SUBDIRECTORY "${PARAVIEW_PLUGIN_SUBDIR}" PLUGINS ${plugins})
if(BUILD_TESTING)
  find_package(Qt6 6.9.0 EXACT REQUIRED COMPONENTS Test)
  add_subdirectory(tests)
endif()
configure_file(cmake/build-info.json.in "${CMAKE_BINARY_DIR}/cae-post-exit-build-info.json" @ONLY)
